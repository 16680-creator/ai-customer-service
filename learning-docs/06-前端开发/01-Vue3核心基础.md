## 二、Vue 3 核心概念

### 2.1 组件 = 模板 + 逻辑 + 样式

```vue
<!-- ChatView.vue 的结构 -->
<template>
  <!-- HTML 模板：定义页面长什么样 -->
  <div class="chat-view">
    <h1>{{ title }}</h1>
  </div>
</template>

<script setup>
// JavaScript 逻辑：定义数据和方法
import { ref } from 'vue'
const title = ref('AI 对话')
</script>

<style scoped>
/* CSS 样式：scoped 表示只对当前组件生效 */
.chat-view { padding: 20px; }
</style>
```

### 2.2 响应式数据（ref / reactive）

```javascript
import { ref, reactive, computed } from 'vue'

// ref：包装基本类型（字符串、数字、布尔）
const count = ref(0)
const message = ref('Hello')
const sending = ref(false)

// 修改值必须用 .value
count.value++
message.value = 'World'

// reactive：包装对象/数组
const state = reactive({
  messages: [],
  currentSession: 'session-1',
  inputMessage: ''
})

// 修改不需要 .value
state.messages.push({ role: 'user', content: '你好' })

// computed：计算属性（自动缓存）
const totalAmount = computed(() => {
  return state.messages.filter(m => m.role === 'user').length
})
```

### 2.3 模板语法

```vue
<template>
  <!-- 文本插值 -->
  <p>{{ message }}</p>
  
  <!-- 条件渲染 -->
  <div v-if="sending">发送中...</div>
  <div v-else>空闲</div>
  
  <!-- 列表渲染 -->
  <div v-for="(msg, index) in messages" :key="index">
    {{ msg.content }}
  </div>
  
  <!-- 事件绑定 -->
  <button @click="sendMessage">发送</button>
  <input @keydown.enter="sendMessage" />
  
  <!-- 双向绑定 -->
  <input v-model="inputMessage" />
  
  <!-- 动态 class -->
  <div :class="['msg-row', msg.role, { active: isActive }]">
  
  <!-- 动态属性 -->
  <img :src="avatarUrl" :alt="userName" />
</template>
```

## 三、本项目的实际代码解析

### 3.1 AI 对话页面（ChatView.vue）

```vue
<script setup>
import { ref, nextTick } from 'vue'
import { chatApi } from '../api'
import { ElMessage } from 'element-plus'

// ===== 响应式数据 =====
const sessions = ref([{ id: 'session-1', title: '默认会话' }])
const currentSession = ref('session-1')
const messages = ref([])        // 聊天消息列表
const inputMessage = ref('')    // 输入框内容
const sending = ref(false)      // 是否正在发送
const messagesRef = ref(null)   // DOM 引用（用于滚动）

// ===== 方法 =====
function newSession() {
  const id = 'session-' + Date.now()
  sessions.value.push({ id, title: '新会话 ' + sessions.value.length })
  currentSession.value = id
  messages.value = []
}

async function sendMessage() {
  const text = inputMessage.value.trim()
  if (!text) return

  // 1. 添加用户消息到列表
  messages.value.push({ role: 'user', content: text })
  inputMessage.value = ''
  sending.value = true
  await scrollToBottom()

  try {
    // 2. 调用后端 API
    const res = await chatApi.post(`/chat/send`, null, {
      params: { sessionId: currentSession.value, message: text }
    })
    // 3. 添加 AI 回复
    const reply = res.data?.data || '未获取到回复'
    messages.value.push({ role: 'assistant', content: reply })
  } catch (e) {
    messages.value.push({ role: 'assistant', content: '❌ 错误: ' + e.message })
    ElMessage.error('对话请求失败')
  } finally {
    sending.value = false
    await scrollToBottom()
  }
}

// 滚动到底部
async function scrollToBottom() {
  await nextTick()  // 等 DOM 更新完
  if (messagesRef.value) {
    messagesRef.value.scrollTop = messagesRef.value.scrollHeight
  }
}
</script>
```

### 3.2 路由配置（router/index.js）

```javascript
import { createRouter, createWebHistory } from 'vue-router'

const routes = [
  { path: '/', name: 'Dashboard', component: () => import('../views/Dashboard.vue') },
  { path: '/chat', name: 'Chat', component: () => import('../views/ChatView.vue') },
  { path: '/user', name: 'User', component: () => import('../views/UserView.vue') },
  { path: '/knowledge', name: 'Knowledge', component: () => import('../views/KnowledgeView.vue') },
  // ...
]

const router = createRouter({
  history: createWebHistory(),  // HTML5 History 模式（无 # 号）
  routes,
})

export default router
```

**`() => import(...)` 是路由懒加载**：只有访问该页面时才加载对应组件，加快首屏速度。

### 3.3 API 封装（api/index.js）

```javascript
import axios from 'axios'

// 各微服务地址
const SERVICES = {
  chat: 'http://localhost:8083',
  user: 'http://localhost:8081',
  knowledge: 'http://localhost:8082',
  order: 'http://localhost:8087',
}

// 创建 Axios 实例
function createClient(baseURL) {
  return axios.create({ baseURL, timeout: 30000 })
}

// 导出各服务的 API 客户端
export const chatApi = createClient(SERVICES.chat)
export const userApi = createClient(SERVICES.user)
export const orderApi = createClient(SERVICES.order)
```

## 五、组件通信

```vue
<!-- 父组件 -->
<template>
  <ChatMessage 
    :message="msg"           <!-- 向下传数据（Props） -->
    @delete="handleDelete"   <!-- 向上发事件（Emit） -->
  />
</template>

<!-- 子组件 ChatMessage.vue -->
<script setup>
// 接收 Props
const props = defineProps({
  message: { type: Object, required: true }
})

// 定义事件
const emit = defineEmits(['delete'])

function onDelete() {
  emit('delete', props.message.id)  // 触发父组件的 handleDelete
}
</script>
```

## 六、常用组合式函数（Composables）

```javascript
// useChat.js - 抽取聊天逻辑
import { ref } from 'vue'
import { chatApi } from '../api'

export function useChat() {
  const messages = ref([])
  const loading = ref(false)

  async function send(sessionId, text) {
    loading.value = true
    try {
      const res = await chatApi.post('/chat/send', null, {
        params: { sessionId, message: text }
      })
      messages.value.push({ role: 'assistant', content: res.data.data })
    } finally {
      loading.value = false
    }
  }

  return { messages, loading, send }
}

// 在组件中使用
// const { messages, loading, send } = useChat()
```
