## 二、`pathlib`：路径操作首选

```python
from pathlib import Path

p = Path("data") / "reports" / "2024.csv"     # 用 / 拼接，跨平台
print(p)                                       # data/reports/2024.csv（Windows 自动反斜杠）

# 属性
p.name        # '2024.csv'
p.stem        # '2024'
p.suffix      # '.csv'
p.parent      # Path('data/reports')
p.parts       # ('data', 'reports', '2024.csv')

# 判断
p.exists(); p.is_file(); p.is_dir()

# 读写（一行）
p.read_text(encoding="utf-8")
p.write_text("x", encoding="utf-8")
p.read_bytes(); p.write_bytes(b"x")

# 目录
Path("out/sub").mkdir(parents=True, exist_ok=True)
p.unlink(missing_ok=True)

# 遍历
for f in Path(".").rglob("*.py"):
    print(f.name, f.stat().st_size)

# 其他
p.resolve()                      # 绝对路径（= toAbsolutePath().normalize()）
p.with_suffix(".json")           # 换扩展名
p.rename(new_path)
```

> **不要再用 `os.path.join` 和字符串拼路径**。`pathlib` 在所有场景都更好，除了极少数需要 `str` 的旧 API（此时 `str(p)`）。

## 四、`json`

```python
import json

# dict → str
```

## 五、`re`：正则

```python
import re

# 基本
re.match(r"^\d+", "123abc")       # 从开头匹配 → Match 或 None
re.search(r"\d+", "abc123")       # 任意位置查找 → Match 或 None
re.fullmatch(r"\d+", "123")       # 全串匹配
re.findall(r"\d+", "a1b22c333")   # ['1','22','333']
re.finditer(r"\d+", "a1b22")      # 迭代 Match 对象（能拿位置）
re.sub(r"\d+", "#", "a1b22")      # 'a#b#'
re.split(r"[,;]", "a,b;c")        # ['a','b','c']

# 编译（重复使用时更快）
PHONE = re.compile(r"^1[3-9]\d{9}$")
if PHONE.match("13800138000"):
    print("合法")

# 提取分组
m = re.search(r"(\d{4})-(\d{2})-(\d{2})", "日期 2024-05-20")
if m:
    print(m.group(0))    # '2024-05-20'
    print(m.group(1))    # '2024'
    print(m.groups())    # ('2024','05','20')
    print(m.groupdict())

# 命名分组（推荐，可读性好）
m = re.search(r"(?P<y>\d{4})-(?P<m>\d{2})", "2024-05")
print(m.group("y"))      # 2024

# 常用 flags
re.findall(r"abc", "ABC", re.IGNORECASE)     # 忽略大小写
re.sub(r"\s+", " ", text)                    # 压空白
re.findall(r"^\w+", text, re.MULTILINE)      # ^ 匹配每行开头
```

### 原始字符串（**必用**）

```python
# ❌ 普通字符串：\b 是退格符，\d 会报警告
re.search("\bword\b", text)

# ✅ 原始字符串：反斜杠字面量
re.search(r"\bword\b", text)
```

> **规则**：正则一律用 `r"..."`。这是 Java 程序员最容易忘的一条（Java 里是 `"\\d"`，Python 是 `r"\d"` 或 `"\\d"`，但 `r` 更清晰）。

## 六、`datetime`

```python
from datetime import datetime, date, time, timedelta, timezone

# 当前
now = datetime.now()                       # 本地时间，无时区（naive）
utc_now = datetime.now(timezone.utc)        # 带时区（aware）
today = date.today()

# 构造
dt = datetime(2024, 5, 20, 14, 30, 0)
dt = datetime.strptime("2024-05-20 14:30", "%Y-%m-%d %H:%M")

# 格式化

dt.strftime("%Y年%m月%d日")             # 中文也行

# 时间戳互转
ts = dt.timestamp()
datetime.fromtimestamp(ts)
datetime.fromtimestamp(ts, tz=timezone.utc)

# 运算
dt + timedelta(days=1, hours=2)
dt - timedelta(weeks=1)
dt2 - dt                                 # → timedelta
(dt2 - dt).total_seconds() / 3600        # 小时数

# 属性
dt.year, dt.month, dt.day, dt.hour, dt.minute, dt.second, dt.weekday()   # 0=周一

datetime.fromisoformat("2024-05-20T14:30:00")

# 时区转换
import zoneinfo                          # 3.9+
```

| `%d` | 两位日 | 20 |

### 坑

```python
# ❌ 混用 aware 和 naive 会 TypeError
datetime.now() - datetime.now(timezone.utc)
# TypeError: can't subtract offset-naive and offset-aware datetimes

# ✅ 统一用 aware（推荐 UTC 存储，展示时转本地）

datetime(2024, 1, 31) + relativedelta(months=1)     # 2024-02-29
```

## 七、`collections`：专用容器

```python
from collections import (
    Counter, defaultdict, deque, OrderedDict, namedtuple, ChainMap
)

# ① Counter：计数（词频统计神器）
c = Counter("hello")
print(c)                        # Counter({'l': 2, 'h': 1, 'e': 1, 'o': 1})
print(c.most_common(2))         # [('l', 2), ('h', 1)]

Counter(["a", "b", "a"]) + Counter(["a"])   # 可加减

# ② defaultdict：免判空的字典（见 03 篇）
d = defaultdict(list)
d["x"].append(1)                # 不存在也不用先建
d2 = defaultdict(int)
d2["k"] += 1                    # 计数

# ③ deque：双端队列（栈 + 队列）
q = deque([1, 2, 3], maxlen=3)
q.append(4)                     # 满了自动挤掉左边 → [2,3,4]
q.appendleft(0)
q.popleft()                     # O(1)，比 list.pop(0) 快
q.rotate(1)                     # 轮转

# ④ namedtuple
Point = namedtuple("Point", ["x", "y"])
p = Point(1, 2)
print(p.x, p[0])

# ⑤ ChainMap：多层配置查找
from collections import ChainMap
cfg = ChainMap(os.environ, defaults)    # 先查环境变量，再查默认值

# ⑥ OrderedDict：3.7+ 后普通 dict 也有序，一般不需要了
#    但 move_to_end / popitem(last=False) 仍有用
```

## 八、`itertools`：迭代器工具

```python
import itertools as it

# 无限
it.count(0, 2)                 # 0,2,4,...
it.cycle([1, 2])               # 1,2,1,2,...
it.repeat("x", 3)              # x,x,x

# 组合
it.chain([1,2], [3,4])         # 1,2,3,4       ← flatMap
it.chain.from_iterable([[1],[2]])  # 同上
it.product([1,2], "ab")        # 笛卡尔积
it.permutations([1,2,3], 2)    # 排列
it.combinations([1,2,3], 2)    # 组合

# 切分
it.islice(naturals, 5)         # 取前 5 个
it.takewhile(lambda x: x < 3, [1,2,3,1])    # [1,2]
it.dropwhile(lambda x: x < 3, [1,2,3,1])    # [3,1]

# 分组（⚠️ 必须先按 key 排序）
data = [("a",1),("a",2),("b",3)]
for key, group in it.groupby(data, key=lambda x: x[0]):
    print(key, list(group))    # a [(a,1),(a,2)] / b [(b,3)]

# 累积
list(it.accumulate([1,2,3,4]))              # [1,3,6,10]
list(it.accumulate([1,2,3], lambda a,b: a*b))  # [1,2,6]

# 成对
list(it.pairwise([1,2,3]))     # [(1,2),(2,3)]   3.10+
```

> `groupby` 的坑：它只对**连续相同 key** 分组。必须 `sorted(data, key=...)` 之后再 groupby，否则同一 key 会被分成多组。Java 的 `Collectors.groupingBy` 没有这个限制，容易踩。

## 九、`functools`

```python
import functools

# 缓存
@functools.lru_cache(maxsize=128)
def fib(n):
    return n if n < 2 else fib(n-1) + fib(n-2)

@functools.cache           # 3.9+ 无大小限制，等价 lru_cache(None)
def heavy(n): ...

# partial：固定部分参数（= 手动柯里化）
from functools import partial
int2 = partial(int, base=2)
int2("1010")               # 10

# reduce
from functools import reduce
reduce(lambda a, b: a + b, [1,2,3,4])     # 10

# 装饰器必备
@functools.wraps(fn)       # 保留元信息（见 04 篇）

# 全序（自动生成 <, <=, >, >=）
@functools.total_ordering
class Money:
    def __eq__(self, o): ...
    def __lt__(self, o): ...
```

> `lru_cache` 是项目里做单例的常用手法（`ai-cs-py-chat` 的 `get_chat_service`）。注意：**被缓存的函数参数必须可哈希**。

## 十、`logging`：别用 print 打日志

```python
import logging

# 最小配置（脚本用）
logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s [%(levelname)s] %(name)s - %(message)s",
    handlers=[
        logging.StreamHandler(),
        logging.FileHandler("app.log", encoding="utf-8"),
    ],
)

logger = logging.getLogger(__name__)      # ← 每个模块这样拿 logger

logger.debug("调试 %s", value)            # 惰性格式化！不要 f-string
logger.info("订单创建 %s 金额 %s", order_no, amount)
logger.warning("库存偏低 %s", sku)
logger.error("调用失败", exc_info=True)    # 带上异常栈
logger.exception("捕获异常")               # 只能在 except 块里用
```

### 为什么用 `%s` 而不是 f-string

```python
logger.debug(f"大对象 {huge_dict}")     # ❌ 即使日志级别是 INFO，也会先构造字符串
logger.debug("大对象 %s", huge_dict)    # ✅ 只有真要输出时才格式化
```

### 与 Java 对照

| Java (SLF4J) | Python logging |
|---|---|
| `LoggerFactory.getLogger(X.class)` | `logging.getLogger(__name__)` |
| `log.info("a={}", a)` | `logger.info("a=%s", a)` |
| `log4j2.xml` | `logging.config.dictConfig` / `basicConfig` |
| `TRACE/DEBUG/INFO/WARN/ERROR` | `DEBUG/INFO/WARNING/ERROR/CRITICAL`（注意 `WARNING` 不是 `WARN`） |
| MDC | `LoggerAdapter` / `contextvars` |
| Appender | Handler |
| PatternLayout | Formatter |

```python
# 生产配置用 dictConfig（等价 logback.xml）
import logging.config

logging.config.dictConfig({
    "version": 1,
    "disable_existing_loggers": False,
    "formatters": {
        "default": {"format": "%(asctime)s [%(levelname)s] %(name)s: %(message)s"},
    },
    "handlers": {
        "console": {"class": "logging.StreamHandler", "formatter": "default"},
        "file": {
            "class": "logging.handlers.RotatingFileHandler",
            "filename": "app.log",
            "maxBytes": 10_000_000,
            "backupCount": 5,
            "encoding": "utf-8",
            "formatter": "default",
        },
    },
    "root": {"level": "INFO", "handlers": ["console", "file"]},
})
```

> **uvicorn 的日志是分开的**：`uvicorn` / `uvicorn.access` 两个 logger，想接管要单独配。

```python
import uuid
import random
```

## 十四、`csv` / `sqlite3`

```python
import csv

# 读（utf-8-sig 兼容 Excel）
with open("data.csv", encoding="utf-8-sig", newline="") as f:
    for row in csv.DictReader(f):
        print(row["姓名"], row["金额"])

# 写（newline="" 必须加，否则 Windows 上多空行）
with open("out.csv", "w", encoding="utf-8-sig", newline="") as f:
    w = csv.DictWriter(f, fieldnames=["姓名", "金额"])
    w.writeheader()
    w.writerow({"姓名": "张三", "金额": 100})
```

```python
import sqlite3

conn = sqlite3.connect("app.db")
conn.row_factory = sqlite3.Row          # 按列名取值

with conn:
    conn.execute("CREATE TABLE IF NOT EXISTS t (id INTEGER PRIMARY KEY, name TEXT)")
    # ⚠️ 参数化查询，永远不要拼字符串（SQL 注入）
    conn.execute("INSERT INTO t (name) VALUES (?)", ("张三",))
    rows = conn.execute("SELECT * FROM t WHERE name = ?", ("张三",)).fetchall()
    for r in rows:
        print(r["id"], r["name"])

conn.close()
```

> **SQL 注入**：`cur.execute(f"... {user_input}")` 是 Python 里最典型的安全漏洞。永远用 `?` 占位 + 参数元组。

## 十五、`math` / `statistics` / `decimal`

```python
import math
import statistics
from decimal import Decimal, ROUND_HALF_UP

# math
math.sqrt(16)          # 4.0
math.ceil(3.2)         # 4
math.floor(3.8)        # 3
math.isclose(0.1+0.2, 0.3)      # True  ← 浮点比较用这个
math.inf, -math.inf
math.pi, math.e

# statistics
statistics.mean([1,2,3])         # 2
statistics.median([1,2,3,4])     # 2.5
statistics.stdev([1,2,3])        # 样本标准差
statistics.quantiles([1,2,3,4,5], n=4)   # 四分位

# decimal（金额必用）
Decimal("0.1") + Decimal("0.2")                     # Decimal('0.3')
(Decimal("100") / Decimal("3")).quantize(Decimal("0.01"), rounding=ROUND_HALF_UP)
```
