## 二、`pathlib`：路径操作首选

```python
# ---- 本段讲：pathlib 路径操作全览（Java 的 Path / Files 对应物）----
# Java 里常见写法是 Paths.get("data", "reports", "2024.csv")，或者字符串拼 "data/" + name；
# Python 用 Path 对象 + 重载的 / 运算符拼路径，跨平台且不会出现漏写分隔符的问题。
from pathlib import Path                       # 导入 Path 类；Java 对应 import java.nio.file.Path

# 连续的 / 构成一条链：每次调用 Path.__truediv__ 都返回新的 Path，原对象不变。
p = Path("data") / "reports" / "2024.csv"     # 用 / 拼接，跨平台
print(p)                                       # data/reports/2024.csv（Windows 自动反斜杠）

# 属性
# 这些是「属性」不是方法：取文件名不用加括号（Java 要写 p.getFileName().toString()）。
p.name        # '2024.csv'   最后一段文件名（含扩展名）
p.stem        # '2024'       去掉最后一个后缀的文件名
p.suffix      # '.csv'       最后一个后缀（带点）
p.parent      # Path('data/reports')   上一级目录
p.parts       # ('data', 'reports', '2024.csv')   拆成路径分量的元组

# 判断
# 分号 ; 可把多条简单语句写进一行（Python 里分号是可选的语句分隔符）。
p.exists(); p.is_file(); p.is_dir()   # 依次判断：存在 / 是文件 / 是目录，返回 bool

# 读写（一行）
# 一行完成「打开 → 读/写 → 关闭」，等价 Java 的 try-with-resources，不会漏关文件。
p.read_text(encoding="utf-8")          # 整个文件读成 str；必须显式指定编码，否则 Windows 用 GBK
p.write_text("x", encoding="utf-8")    # 覆盖写入字符串；文件不存在会自动创建
p.read_bytes(); p.write_bytes(b"x")    # 二进制版本：bytes 字面量前缀 b，对应 Java 的 byte[]

# 目录
# parents=True = mkdir -p（父目录一起建）；exist_ok=True = 目录已存在也不抛 FileExistsError。
Path("out/sub").mkdir(parents=True, exist_ok=True)
p.unlink(missing_ok=True)              # 删除文件；missing_ok=True 时文件不存在也不报错

# 遍历
# rglob("*.py") 是递归匹配（等价 glob("**/*.py")），返回生成器，for 循环时才逐个产出路径。
for f in Path(".").rglob("*.py"):
    print(f.name, f.stat().st_size)     # stat() 取文件元数据；st_size 是字节数，对应 Files.size()

# 其他
p.resolve()                      # 绝对路径（= toAbsolutePath().normalize()）
p.with_suffix(".json")           # 换扩展名
p.rename(new_path)               # 重命名/移动；new_path 在片段中被省略，实际需先定义
```

> **不要再用 `os.path.join` 和字符串拼路径**。`pathlib` 在所有场景都更好，除了极少数需要 `str` 的旧 API（此时 `str(p)`）。

## 三、`os` / `sys`：系统和运行时

```python
# ---- 本段讲：os / sys——环境变量、工作目录、解释器信息、退出码 ----
# Java 的 System.getenv / System.getProperty / System.exit 在 Python 被拆成两个模块：
# os 管操作系统资源，sys 管解释器自身（版本、搜索路径、标准流）。
import os                                    # 系统层：环境变量、cwd、平台标识
import sys                                   # 解释器层：版本、executable、sys.path、argv

# 环境变量
# os.environ 是「类字典」对象，读写都映射到进程环境；Java 对应 System.getenv()。
os.environ.get("LLM_API_KEY")              # 不存在返回 None（推荐）
os.environ["LLM_API_KEY"]                  # 不存在抛 KeyError
os.environ.get("PORT", "8000")             # 第二个参数是默认值：不存在时返回 "8000"（仍是 str，要自己 int()）
os.environ["NEW"] = "v"                    # 仅本进程有效

# 工作目录（注意：影响相对路径！）
# getcwd = 当前工作目录（不是脚本所在目录）；chdir 会改变后续所有相对路径的基准。
os.getcwd()                                # 返回当前工作目录的绝对路径字符串
os.chdir("../")                            # 切换到上一级目录；出错抛 FileNotFoundError

# 平台判断
sys.platform          # 'win32' / 'linux' / 'darwin'
os.name               # 'nt' / 'posix'（更粗粒度；Windows 是 nt）

# 解释器信息
sys.version           # 完整版本字符串，含编译信息
sys.executable        # python.exe 的绝对路径（排查环境问题必备）
sys.path              # 模块搜索路径
sys.argv              # 命令行参数（argv[0] 是脚本名）

# 退出
sys.exit(1)           # 非 0 = 异常退出（对应 System.exit）
sys.exit("错误信息")   # 打印到 stderr 并退出 1

# 标准流
# print 的 file= 参数决定输出目标；默认就是 sys.stdout，这里显式写出来做对照。
print("out", file=sys.stdout)     # 标准输出，正常结果走这里
print("err", file=sys.stderr)     # 标准错误，错误/告警走这里（可被单独重定向）
sys.stdin.readline()              # 读一行标准输入（含结尾换行符），对应 Java 的 BufferedReader.readLine()
```

### 加载 `.env`（项目用法）

```python
# ---- 本段讲：把 .env 文件加载成配置的两种方式 ----
# .env 是「KEY=VALUE」的纯文本，放本地密钥，绝不能提交到 git；
# Java 里对应 Spring Boot 的 application.properties + @Value / @ConfigurationProperties。
# 方式一：python-dotenv
# 第三方包（需 pip 安装）；作用是把 .env 的每条键值塞进 os.environ。
from dotenv import load_dotenv
load_dotenv()                    # 读 .env 写入 os.environ
print(os.environ["LLM_API_KEY"])  # 现在可以直接按下标取；若 .env 里没有仍会 KeyError

# 方式二：pydantic-settings（项目实际用法，带类型校验）
# 思路类似 Spring 的 @ConfigurationProperties：声明字段即完成「读取 + 类型转换 + 校验」。
from pydantic_settings import BaseSettings, SettingsConfigDict

class Settings(BaseSettings):                     # 继承 BaseSettings，框架自动从环境变量/.env 填充字段
    model_config = SettingsConfigDict(env_file=".env", extra="ignore")  # 指定 .env 路径；extra="ignore" 忽略多余键
    llm_api_key: str = ""                         # 类型注解 str + 默认值；环境变量 LLM_API_KEY 会自动映射到它
    llm_base_url: str = "https://api.deepseek.com"  # 同上，缺失时用默认值而不是报错

settings = Settings()                             # 实例化时才真正读取 .env 并做校验；字段类型不对会在这里抛错
```

## 四、`json`

```python
# ---- 本段讲：json 模块——对象与字符串/文件互转 ----
# Python 的 dict/list 和 JSON 结构几乎一一对应，所以序列化只需一行；
# Java 里要 ObjectMapper.writeValueAsString / readValue（Jackson）。
import json

# dict → str
# dumps = dump string；ensure_ascii=False 让中文原样输出，否则会变成 \u540d 这种转义。
json.dumps({"名": "中文"}, ensure_ascii=False, indent=2)   # indent=2 表示缩进两个空格美化输出
# '{\n  "名": "中文"\n}'

# str → dict
# loads = load string；返回的是 dict（嵌套对象也是 dict，嵌套数组是 list）。
json.loads('{"a": 1}')

# 文件
# 带 s 的是字符串版本，不带的 dump/load 直接读写文件对象；with 保证自动 close。
with open("a.json", "w", encoding="utf-8") as f:
    json.dump(data, f, ensure_ascii=False, indent=2)   # data 需在前面定义；写入时同样别忘 ensure_ascii

with open("a.json", encoding="utf-8") as f:            # 读文件默认模式是 "r"，可省略
    data = json.load(f)                                # 反序列化成 Python 对象

# 常用参数
json.dumps(data, sort_keys=True)         # 键排序，便于 diff
json.dumps(data, separators=(",", ":"))  # 紧凑（省流量）
json.dumps(data, default=str)            # 不可序列化对象转 str（兜底）
json.loads(s, parse_float=Decimal)       # 金额精度

# 异常
# JSONDecodeError 是 ValueError 的子类；msg/lineno/colno 能直接定位坏在哪。
try:
    json.loads("{bad}")
except json.JSONDecodeError as e:
    print(e.msg, e.lineno, e.colno)      # msg=错误描述，lineno=第几行，colno=第几列
```

> **必记**：`ensure_ascii=False`。否则中文全变 `\uXXXX`，可读性归零。

## 五、`re`：正则

```python
# ---- 本段讲：re 正则的常用 API（match/search/findall/sub/split/分组）----
# 与 Java 的 java.util.regex 概念一致，但 Python 的「模块级函数」不用先 Pattern.compile；
# 另外正则字符串一律用 r"..."（原始字符串），否则 \d 会被 Python 字符串层先解释一遍。
import re

# 基本
# match 只从字符串开头尝试；search 扫描整串；fullmatch 要求整串完全匹配。
re.match(r"^\d+", "123abc")       # 从开头匹配 → Match 或 None
re.search(r"\d+", "abc123")       # 任意位置查找 → Match 或 None
re.fullmatch(r"\d+", "123")       # 全串匹配
re.findall(r"\d+", "a1b22c333")   # ['1','22','333']  直接返回字符串列表
re.finditer(r"\d+", "a1b22")      # 迭代 Match 对象（能拿位置）
re.sub(r"\d+", "#", "a1b22")      # 'a#b#'  替换，对应 replaceAll
re.split(r"[,;]", "a,b;c")        # ['a','b','c']  按正则切分

# 编译（重复使用时更快）
# 把「正则文本 → 自动机」的编译结果缓存到变量里，循环中复用；Java 的 Pattern 也是这个用途。
PHONE = re.compile(r"^1[3-9]\d{9}$")   # 中国大陆手机号：1 开头 + 3~9 + 9 位数字
if PHONE.match("13800138000"):          # 用编译后的对象调用，语义同 re.match
    print("合法")

# 提取分组
# 圆括号 () 是捕获组，group(n) 取第 n 组；group(0) 是整段匹配。
m = re.search(r"(\d{4})-(\d{2})-(\d{2})", "日期 2024-05-20")
if m:                                     # 匹配失败时 m 是 None，必须先判空再取 group
    print(m.group(0))    # '2024-05-20'
    print(m.group(1))    # '2024'
    print(m.groups())    # ('2024','05','20')
    print(m.groupdict()) # {} 没有命名组时是空字典

# 命名分组（推荐，可读性好）
# (?P<名字>...) 给组起名；Java 的写法是 (?<名字>...)，少一个 P。
m = re.search(r"(?P<y>\d{4})-(?P<m>\d{2})", "2024-05")
print(m.group("y"))      # 2024

# 常用 flags
re.findall(r"abc", "ABC", re.IGNORECASE)     # 忽略大小写
re.sub(r"\s+", " ", text)                    # 压空白；text 为片段占位，需自行定义
re.findall(r"^\w+", text, re.MULTILINE)      # ^ 匹配每行开头；不加 MULTILINE 时 ^ 只匹配整串开头
```

### 原始字符串（**必用**）

```python
# ---- 本段讲：正则必须用原始字符串 r"..."，对比普通字符串的错误 ----
# 普通字符串里反斜杠会先被 Python 的字符串转义处理，正则引擎拿到的已经不是你想要的东西。
# ❌ 普通字符串：\b 是退格符，\d 会报警告
# "\b" 在字符串层就变成了 ASCII 退格字符（\x08），正则里再也表示不了「单词边界」。
re.search("\bword\b", text)        # text 为片段占位，需自行定义；这行匹配的不是「单词 word」

# ✅ 原始字符串：反斜杠字面量
# 前缀 r 表示「不处理转义」，反斜杠原样传给正则引擎；Java 里没有 r 前缀，只能写 "\\bword\\b"。
re.search(r"\bword\b", text)       # 现在 \b 才是正则的单词边界
```

> **规则**：正则一律用 `r"..."`。这是 Java 程序员最容易忘的一条（Java 里是 `"\\d"`，Python 是 `r"\d"` 或 `"\\d"`，但 `r` 更清晰）。

## 六、`datetime`

```python
# ---- 本段讲：datetime 的获取、构造、格式化、时间戳、运算、时区 ----
# 整体对应 Java 的 java.time（LocalDateTime / ZonedDateTime / Duration）；
# 最大的差异：Python 的 datetime 分「naive（无时区）」和「aware（带时区）」两种，混用会报错。
from datetime import datetime, date, time, timedelta, timezone   # 常用类一次导入；time 这里是「时刻」不是 time 模块

# 当前
now = datetime.now()                       # 本地时间，无时区（naive）
utc_now = datetime.now(timezone.utc)        # 带时区（aware）
today = date.today()                       # 只要日期，返回 date 对象（没有时分秒）

# 构造
# 直接给年月日时分秒，等价 Java 的 LocalDateTime.of(2024, 5, 20, 14, 30, 0)。
dt = datetime(2024, 5, 20, 14, 30, 0)
dt = datetime.strptime("2024-05-20 14:30", "%Y-%m-%d %H:%M")   # 字符串 → datetime，格式符要与输入完全对应

# 格式化
# strftime = string format time；%Y 四位年、%m 月、%d 日、%H:%M:%S 时分秒。
dt.strftime("%Y-%m-%d %H:%M:%S")      # '2024-05-20 14:30:00'
dt.strftime("%Y年%m月%d日")             # 中文也行

# 时间戳互转
# timestamp() 返回自 1970-01-01 UTC 起的秒数（float，带小数毫秒），对应 Java 的 Instant.toEpochMilli()/1000。
ts = dt.timestamp()
datetime.fromtimestamp(ts)                # 按「本地时区」解释，返回 naive datetime
datetime.fromtimestamp(ts, tz=timezone.utc)   # 显式指定 UTC，返回 aware datetime（推荐）

# 运算
# timedelta 是「时间间隔」，对应 Java 的 Duration / Period；日期 + 间隔得到新对象（原对象不可变）。
dt + timedelta(days=1, hours=2)           # 加 1 天 2 小时
dt - timedelta(weeks=1)                   # 减 1 周（weeks=1 等价 days=7）
dt2 - dt                                 # → timedelta
(dt2 - dt).total_seconds() / 3600        # 小时数；dt2 为片段占位，需自行定义

# 属性
# 全部是属性访问，不用 getYear() 这类方法；weekday() 是方法，0 表示周一（Java 的 DayOfWeek 是 1=周一）。
dt.year, dt.month, dt.day, dt.hour, dt.minute, dt.second, dt.weekday()   # 0=周一

# ISO 格式（最省事，跨语言通用）
# ISO 8601 形如 2024-05-20T14:30:00；前后端、数据库、日志都认，优先用它而不是自定义格式。
dt.isoformat()                           # '2024-05-20T14:30:00'
datetime.fromisoformat("2024-05-20T14:30:00")   # ISO 字符串 → datetime，不用写格式符

# 时区转换
# zoneinfo 是 3.9+ 标准库（等价 Java 的 ZoneId）；astimezone 只改表示，不改变时间点。
import zoneinfo                          # 3.9+
cn = zoneinfo.ZoneInfo("Asia/Shanghai")   # 时区 ID 用「大洲/城市」格式
utc_dt.astimezone(cn)                     # 把 UTC 时间转成上海时间显示
```

### 常见格式符

| `%d` | 两位日 | 20 |
| `%H` `%M` `%S` | 时分秒（24h） | 14:30:00 |

### 坑

```python
# ---- 本段讲：datetime 三个高频坑——naive/aware 混用、没有 Period、时区 ----
# ❌ 混用 aware 和 naive 会 TypeError
# 左边 datetime.now() 不带时区，右边带 UTC，Python 拒绝猜测，直接抛异常。
datetime.now() - datetime.now(timezone.utc)   # 两个操作数一个 naive 一个 aware，无法相减
# TypeError: can't subtract offset-naive and offset-aware datetimes

# ✅ 统一用 aware（推荐 UTC 存储，展示时转本地）
# 存库统一 UTC（带 tz），只在展示给用户时 astimezone() 成本地时区，避免跨时区错乱。
now_utc = datetime.now(timezone.utc)

# ⚠️ Python 的 datetime 不支持"年月加减"（没有 Period）
# timedelta 只有天/秒等单位，加 1 个月没法用 timedelta 表达（每月天数不同）。
# 需要的话用第三方 dateutil
from dateutil.relativedelta import relativedelta    # 第三方
datetime(2024, 1, 31) + relativedelta(months=1)     # 2024-02-29  自动处理月末，超出则取当月最后一天
```

## 七、`collections`：专用容器

```python
# ---- 本段讲：collections 的六个专用容器（Counter/defaultdict/deque/namedtuple/ChainMap/OrderedDict）----
# 这些是「针对特定场景优化的 dict/list 子类」，Java 里往往要引 Guava 或自己写；
# Python 直接把最高频的几种装进了标准库。
from collections import (
    Counter, defaultdict, deque, OrderedDict, namedtuple, ChainMap
)   # 一行导入多个名字；括号只是为了让长 import 换行，语法等价于逗号连接

# ① Counter：计数（词频统计神器）
# Counter 是 dict 的子类：键=元素，值=出现次数；传入可迭代对象会自动统计。
c = Counter("hello")             # 字符串也是可迭代对象，逐字符计数
print(c)                        # Counter({'l': 2, 'h': 1, 'e': 1, 'o': 1})
print(c.most_common(2))         # [('l', 2), ('h', 1)]  按次数降序取前 2

Counter(["a", "b", "a"]) + Counter(["a"])   # 可加减；同键计数相加，Counter({'a': 3, 'b': 1})

# ② defaultdict：免判空的字典（见 03 篇）
# 访问不存在的键时，自动用「工厂函数」造一个默认值并写回字典，省掉 if key not in d。
d = defaultdict(list)            # 工厂是 list，所以缺失的键自动得到 []
d["x"].append(1)                # 不存在也不用先建
d2 = defaultdict(int)            # 工厂是 int，缺失的键自动得到 0
d2["k"] += 1                    # 计数；等价于 d2["k"] = d2["k"] + 1

# ③ deque：双端队列（栈 + 队列）
# deque = double-ended queue，两端插入/删除都是 O(1)；list 头部插入是 O(n)。
q = deque([1, 2, 3], maxlen=3)   # maxlen 限制容量，超了自动从另一端丢弃
q.append(4)                     # 满了自动挤掉左边 → [2,3,4]
q.appendleft(0)                 # 左端插入，O(1)
q.popleft()                     # O(1)，比 list.pop(0) 快
q.rotate(1)                     # 轮转：所有元素右移一位，最右的移到最左

# ④ namedtuple
# 用「字段名」访问的轻量不可变元组；类似 Java 的 record，但运行时动态创建。
Point = namedtuple("Point", ["x", "y"])   # 第一个参数是类型名，第二个是字段名列表
p = Point(1, 2)                           # 按位置传参，生成不可变对象
print(p.x, p[0])                          # 1 1：既可按字段名，也可按下标访问

# ⑤ ChainMap：多层配置查找
# 把多个 dict 串成「逻辑上的一个」，查找时从前往后命中即返回，写操作只落到第一个。
from collections import ChainMap
cfg = ChainMap(os.environ, defaults)    # 先查环境变量，再查默认值；defaults 为片段占位，需自行定义

# ⑥ OrderedDict：3.7+ 后普通 dict 也有序，一般不需要了
#    但 move_to_end / popitem(last=False) 仍有用；需要「把某键移到末尾」或「从头部弹出」时才用
```

## 八、`itertools`：迭代器工具

```python
# ---- 本段讲：itertools——惰性迭代器工具（无限序列、组合、切分、分组、累积）----
# 所有函数都返回「迭代器」而不是列表，用多少算多少，内存 O(1)；
# Java 里部分能力要 Stream.generate / flatMap / Collectors 才能拼出来。
import itertools as it          # 起别名 it，少打字；这是社区惯例

# 无限
# 无限迭代器本身不会停，必须配合 islice / zip / break 使用，否则会死循环。
it.count(0, 2)                 # 0,2,4,...   等差数列，默认步长 1
it.cycle([1, 2])               # 1,2,1,2,... 循环重复，序列耗尽后从头再来
it.repeat("x", 3)              # x,x,x       重复固定值，给了次数就是有限个

# 组合
# chain 把多个可迭代对象串成一条，对应 Java Stream 的 flatMap。
it.chain([1,2], [3,4])         # 1,2,3,4       ← flatMap
it.chain.from_iterable([[1],[2]])  # 同上；参数是「装着多个可迭代对象的可迭代对象」
it.product([1,2], "ab")        # 笛卡尔积：(1,'a'),(1,'b'),(2,'a'),(2,'b')
it.permutations([1,2,3], 2)    # 排列：顺序有关，(1,2) 和 (2,1) 都算
it.combinations([1,2,3], 2)    # 组合：顺序无关，只出 (1,2),(1,3),(2,3)

# 切分
# islice 是「迭代器版切片」，惰性取前 n 个，不会先把整个序列算出来。
it.islice(naturals, 5)         # 取前 5 个；naturals 为片段占位，代表任意无限迭代器
it.takewhile(lambda x: x < 3, [1,2,3,1])    # [1,2]  条件一旦为假就停（后面的 1 不会再取）
it.dropwhile(lambda x: x < 3, [1,2,3,1])    # [3,1]  跳过开头满足条件的元素，遇到不满足就全部保留

# 分组（⚠️ 必须先按 key 排序）
# groupby 只比较「相邻」元素，所以相同 key 必须连续；这点和 Java 的 groupingBy 完全不同。
data = [("a",1),("a",2),("b",3)]              # data 已按 key 排好（a,a,b）
for key, group in it.groupby(data, key=lambda x: x[0]):   # key 决定按什么分组
    print(key, list(group))    # a [(a,1),(a,2)] / b [(b,3)]  group 是迭代器，必须 list() 才能看内容

# 累积
# accumulate 把「前一步结果」和「当前元素」做二元运算，类似 reduce 但保留每一步的结果。
list(it.accumulate([1,2,3,4]))              # [1,3,6,10]  默认加法：1, 1+2, 3+3, 6+4
list(it.accumulate([1,2,3], lambda a,b: a*b))  # [1,2,6]  自定义运算：累乘

# 成对
# pairwise 把序列变成相邻两两成对的迭代器，处理「前后比较」很方便。
list(it.pairwise([1,2,3]))     # [(1,2),(2,3)]   3.10+
```

> `groupby` 的坑：它只对**连续相同 key** 分组。必须 `sorted(data, key=...)` 之后再 groupby，否则同一 key 会被分成多组。Java 的 `Collectors.groupingBy` 没有这个限制，容易踩。

## 九、`functools`

```python
# ---- 本段讲：functools——缓存、偏函数、reduce、装饰器工具、全序 ----
# Java 没有直接对应物：缓存像 Spring 的 @Cacheable，偏函数像手动柯里化，
# total_ordering 类似 Comparable 的自动补全。
import functools               # 导入整个模块，后面用 functools.xxx 前缀调用

# 缓存
# lru_cache 把「参数 → 返回值」存进字典：同一参数第二次调用直接命中，不再执行函数体。
# 参数必须可哈希（list/dict 不行），因为他要用参数元组做 key。
@functools.lru_cache(maxsize=128)   # 装饰器；maxsize=128 表示最多缓存 128 组结果，超出按 LRU 淘汰
def fib(n):                          # 朴素递归斐波那契，不加缓存是指数级复杂度
    return n if n < 2 else fib(n-1) + fib(n-2)   # 单行三元表达式；递归调用同样走缓存

@functools.cache           # 3.9+ 无大小限制，等价 lru_cache(None)
def heavy(n): ...          # 函数体省略（... 是占位符，合法语法）；实际应返回计算结果

# partial：固定部分参数（= 手动柯里化）
# partial(int, base=2) 生成一个新函数：调用时只需再传字符串，base 已固定为 2。
from functools import partial
int2 = partial(int, base=2)   # 相当于 lambda s: int(s, base=2)
int2("1010")               # 10  二进制 1010 转十进制

# reduce
# reduce 用二元函数把序列「折叠」成一个值；Python 3 把它挪进了 functools（内置只剩 sum/max 等特例）。
from functools import reduce
reduce(lambda a, b: a + b, [1,2,3,4])     # 10  过程：((1+2)+3)+4；初始值省略时用第一个元素

# 装饰器必备
# 写装饰器时套在 wrapper 上，把原函数的 __name__/__doc__/签名复制回来（见 04 篇）。
@functools.wraps(fn)       # 保留元信息（见 04 篇）；fn 为片段占位，代表被包装的原函数

# 全序（自动生成 <, <=, >, >=）
# 只需实现 __eq__ 和 __lt__，其余比较运算符由装饰器推导，省掉大量样板代码。
@functools.total_ordering
class Money:
    def __eq__(self, o): ...     # 判断相等；o 是另一个 Money 实例
    def __lt__(self, o): ...     # 判断小于；total_ordering 据此推导 <=、>、>=
```

> `lru_cache` 是项目里做单例的常用手法（`ai-cs-py-chat` 的 `get_chat_service`）。注意：**被缓存的函数参数必须可哈希**。

## 十、`logging`：别用 print 打日志

```python
# ---- 本段讲：logging 的最小可用配置与五个级别 ----
# 对应 SLF4J + Logback：basicConfig 相当于给 root logger 装 Handler 和 Formatter，
# getLogger(__name__) 等价 LoggerFactory.getLogger(X.class)。
import logging

# 最小配置（脚本用）
# 只在程序启动时调用一次；重复调用不会覆盖已有配置（basicConfig 第二次是空操作）。
logging.basicConfig(
    level=logging.INFO,                     # 全局最低级别：低于 INFO 的 DEBUG 不输出
    format="%(asctime)s [%(levelname)s] %(name)s - %(message)s",   # 时间 [级别] logger名 - 消息
    handlers=[                              # 同时输出到控制台和文件
        logging.StreamHandler(),            # 控制台 Handler，默认写 stderr
        logging.FileHandler("app.log", encoding="utf-8"),   # 文件 Handler；中文必须指定 utf-8
    ],
)

logger = logging.getLogger(__name__)      # ← 每个模块这样拿 logger

# 注意：下面用 %s 占位 + 参数分离的写法，字符串只在「真要输出」时才拼接（惰性格式化）。
logger.debug("调试 %s", value)            # 惰性格式化！不要 f-string；value 为片段占位
logger.info("订单创建 %s 金额 %s", order_no, amount)   # 多个占位按顺序对应后面的参数
logger.warning("库存偏低 %s", sku)        # 注意级别名是 WARNING 不是 WARN
logger.error("调用失败", exc_info=True)    # 带上异常栈；exc_info=True 会附上当前异常 traceback
logger.exception("捕获异常")               # 只能在 except 块里用；等价 error(..., exc_info=True)
```

### 为什么用 `%s` 而不是 f-string

```python
# ---- 本段讲：为什么日志参数用 %s 占位，而不是 f-string ----
# f-string 是「立即求值」：无论日志级别开没开，字符串都会先被拼出来，白白浪费 CPU 和内存。
# %s 占位是「延迟求值」：logging 内部先判断级别，只有真要输出时才做格式化。
logger.debug(f"大对象 {huge_dict}")     # ❌ 即使日志级别是 INFO，也会先构造字符串
logger.debug("大对象 %s", huge_dict)    # ✅ 只有真要输出时才格式化
```

### 与 Java 对照

| Java (SLF4J) | Python logging |
|---|---|
| `LoggerFactory.getLogger(X.class)` | `logging.getLogger(__name__)` |
| `log.info("a={}", a)` | `logger.info("a=%s", a)` |
| `log4j2.xml` | `logging.config.dictConfig` / `basicConfig` |
| `TRACE/DEBUG/INFO/WARN/ERROR` | `DEBUG/INFO/WARNING/ERROR/CRITICAL`（注意 `WARNING` 不是 `WARN`） |
| MDC | `LoggerAdapter` / `contextvars` |
| Appender | Handler |
| PatternLayout | Formatter |

```python
# ---- 本段讲：用 dictConfig 做生产级日志配置（等价 logback.xml）----
# 用一个嵌套 dict 描述 formatter / handler / logger 三层结构，比 basicConfig 灵活得多；
# Java 对应把 logback.xml 换成代码或 YAML 配置。
# 生产配置用 dictConfig（等价 logback.xml）
import logging.config           # dictConfig 在这个子模块里，必须先 import

logging.config.dictConfig({                      # 整个 dict 就是一份「日志配置树」
    "version": 1,                                # 配置格式版本，目前固定写 1
    "disable_existing_loggers": False,           # 不关闭已存在的 logger（避免第三方库日志突然消失）
    "formatters": {                              # 定义格式器，可被 handler 按名字引用
        "default": {"format": "%(asctime)s [%(levelname)s] %(name)s: %(message)s"},
    },
    "handlers": {                                # 定义输出目标（Appender）
        "console": {"class": "logging.StreamHandler", "formatter": "default"},   # 控制台
        "file": {                                # 文件处理器：按大小滚动切割，防止单个文件无限增长
            "class": "logging.handlers.RotatingFileHandler",   # 滚动文件 Handler 的完整类路径
            "filename": "app.log",               # 日志文件路径（相对当前工作目录）
            "maxBytes": 10_000_000,              # 单文件上限 10MB；数字下划线只是可读性分隔，等价 10000000
            "backupCount": 5,                    # 最多保留 5 个历史文件，再旧的就删除
            "encoding": "utf-8",                 # 中文日志必须显式指定编码
            "formatter": "default",              # 引用上面定义的格式器
        },
    },
    "root": {"level": "INFO", "handlers": ["console", "file"]},   # 根 logger：级别 + 挂哪些 handler
})
```

> **uvicorn 的日志是分开的**：`uvicorn` / `uvicorn.access` 两个 logger，想接管要单独配。

## 十一、`subprocess`：执行外部命令

```python
# ---- 本段讲：subprocess 执行外部命令（run 一次性 / Popen 流式）----
# 对应 Java 的 ProcessBuilder + Process；列表参数形式不会经过 shell，天然避免命令注入。
import subprocess

# 推荐：run + 列表参数（不经 shell，无注入风险）
# 每个参数单独一项，Windows 上也不会出现空格/引号被拆错的问题。
r = subprocess.run(
    ["git", "status", "--short"],   # 列表第一项是可执行文件名，后面依次是参数
    capture_output=True,            # 捕获 stdout/stderr 到返回值，不直接打印到终端
    text=True,                # 返回 str 而非 bytes
    encoding="utf-8",         # Windows 务必指定
    timeout=30,               # 超时（秒）；超时抛 TimeoutExpired，避免子进程卡死主程序
    check=False,              # True = 非 0 退出码抛异常
)
print(r.returncode)               # 退出码：0 表示成功，非 0 表示失败
print(r.stdout)                   # 标准输出（已解码成 str）
print(r.stderr)                   # 标准错误（已解码成 str）

# 需要 shell 特性时（⚠️ 有注入风险，别拼用户输入）
# shell=True 会让命令交给 cmd.exe/bash 解释，管道、通配符等才有意义，但用户输入可被注入。
subprocess.run("dir /b", shell=True, capture_output=True, text=True, encoding="utf-8")

# 交互式/流式读取
# Popen 不会等命令结束，可以边跑边读，适合长时间运行、输出很多的进程（如 ping、构建）。
with subprocess.Popen(["ping", "-n", "3", "127.0.0.1"], stdout=subprocess.PIPE,
                      text=True, encoding="utf-8") as p:   # with 结束时自动关闭管道并等待进程
    for line in p.stdout:         # 逐行迭代子进程 stdout，避免一次性读入内存
        print(line.rstrip())      # rstrip 去掉行尾换行，避免 print 再补一个变成空行
```

```java
// Java 等价
Process p = new ProcessBuilder("git", "status", "--short").start();
```

> **Windows 坑**：不指定 `encoding="utf-8"` 且系统是 GBK 时，中文输出会 `UnicodeDecodeError`。统一加 `encoding="utf-8", errors="replace"`。

## 十二、`argparse`：命令行参数

```python
# ---- 本段讲：argparse 定义命令行参数（Java 的 picocli / Commons CLI）----
# 只做三件事：建 parser → 逐个声明参数 → parse_args() 拿到命名空间对象。
import argparse

        print(f"参数：{args}")                      # Namespace 的字符串形式，调试时看所有参数很方便

if __name__ == "__main__":                         # 仅当脚本被直接运行时才执行；被 import 时不跑
    main()
```

## 十三、`hashlib` / `uuid` / `random` / `secrets`

```python
# ---- 本段讲：哈希、UUID、伪随机、安全随机四组工具 ----
# 哈希 = MessageDigest；uuid4 = UUID.randomUUID()；random 类似 java.util.Random；
# secrets 专门用于不可预测的随机（密码重置 token、会话 ID），底层用操作系统熵源。
import hashlib
import uuid
import random
import secrets

# 哈希
# 参数必须是 bytes，不能直接传 str（字符串要先 .encode()）；hexdigest 返回十六进制串。
hashlib.md5(b"data").hexdigest()               # MD5 已不安全，只适合做校验和，不要用于密码
hashlib.sha256(b"data").hexdigest()            # SHA-256，通用摘要算法
hashlib.sha256("中文".encode("utf-8")).hexdigest()   # 中文先编码成 UTF-8 字节再哈希

# 大文件分段哈希
# 一次读 8KB，边读边 update，内存占用恒定；等价 Java 的 DigestInputStream。
h = hashlib.sha256()                           # 先创建「空」的哈希对象
with open("big.bin", "rb") as f:               # "rb" = 二进制只读模式，避免文本解码
    for chunk in iter(lambda: f.read(8192), b""):   # 反复读 8KB，直到读出 b""（EOF）；iter 的哨兵写法
        h.update(chunk)                        # 把这一段字节喂给哈希对象，可多次调用累加
print(h.hexdigest())                           # 全部读完后一次性取十六进制结果

# UUID
uuid.uuid4()                       # 随机（= UUID.randomUUID()）
uuid.uuid1()                       # 基于时间 + MAC
str(uuid.uuid4())                  # 转成带连字符的字符串，写 JSON/日志时用

# 随机（伪随机，不要用于安全场景）
# 伪随机由种子决定，可被预测；用于抽奖、采样、打乱顺序可以，用于 token 绝对不行。
random.random()                    # [0, 1)  浮点数
random.randint(1, 10)              # 闭区间：两端都可能取到（Python 与 Java 的 Random 语义一致）
random.choice(["a", "b"])          # 从序列里等概率挑一个
random.sample(range(100), 5)       # 不重复抽样：返回 5 个互不相同的元素
random.shuffle(lst)                # 原地打乱；返回 None（不是新列表）
random.seed(42)                    # 可复现（测试用）：固定种子后每次随机序列都相同

# 安全随机（密码、token）
# secrets 用操作系统的密码学安全随机源，不可预测；Java 对应 SecureRandom。
secrets.token_hex(16)              # 32 位十六进制
secrets.token_urlsafe(32)          # URL 安全：base64url 编码，可直接放链接参数
secrets.choice("abcdef")           # 安全地从字符集中选一个字符
```

## 十四、`csv` / `sqlite3`

```python
# ---- 本段讲：csv 模块按「字典」读写，处理表头和 Windows 换行 ----
# Java 常用 OpenCSV / Apache Commons CSV；Python 标准库直接支持 DictReader / DictWriter。
import csv

# 读（utf-8-sig 兼容 Excel）
# utf-8-sig 会自动吃掉 BOM 头（Excel 存 UTF-8 CSV 时会在文件开头加 \ufeff）；
# newline="" 交给 csv 模块自己处理换行，避免行被拆错。
with open("data.csv", encoding="utf-8-sig", newline="") as f:
    for row in csv.DictReader(f):      # 第一行自动当表头，每行返回 {列名: 值} 的 dict
        print(row["姓名"], row["金额"])  # 按列名取值；值都是 str，需要数字要自己 float()/int()

# 写（newline="" 必须加，否则 Windows 上多空行）
# 写时用 utf-8-sig 是为了让 Excel 双击打开不乱码。
with open("out.csv", "w", encoding="utf-8-sig", newline="") as f:
    w = csv.DictWriter(f, fieldnames=["姓名", "金额"])   # 先声明列顺序，决定表头和每行字段顺序
    w.writeheader()                                      # 写表头行
    w.writerow({"姓名": "张三", "金额": 100})             # 写一行；dict 的键要覆盖 fieldnames 里的所有列
```

```python
# ---- 本段讲：sqlite3——零配置嵌入式数据库，以及参数化查询防注入 ----
# 对应 JDBC + H2：connect 拿连接，execute 执行 SQL，? 是占位符（Java 的 PreparedStatement）。
import sqlite3

conn = sqlite3.connect("app.db")        # 文件不存在会自动创建；传 ":memory:" 则是内存库
conn.row_factory = sqlite3.Row          # 按列名取值

# with conn 不是关闭连接，而是「事务上下文」：正常结束自动 commit，异常自动 rollback。
with conn:
    conn.execute("CREATE TABLE IF NOT EXISTS t (id INTEGER PRIMARY KEY, name TEXT)")   # 建表；IF NOT EXISTS 保证可重复执行
    # ⚠️ 参数化查询，永远不要拼字符串（SQL 注入）
    conn.execute("INSERT INTO t (name) VALUES (?)", ("张三",))   # ? 占位，第二个参数必须是元组（单元素也要逗号）
    rows = conn.execute("SELECT * FROM t WHERE name = ?", ("张三",)).fetchall()   # fetchall 取全部结果行
    for r in rows:
        print(r["id"], r["name"])       # 因为设置了 row_factory，可以像 dict 一样按列名取

conn.close()                            # 显式关闭连接，释放文件句柄
```

> **SQL 注入**：`cur.execute(f"... {user_input}")` 是 Python 里最典型的安全漏洞。永远用 `?` 占位 + 参数元组。

## 十五、`math` / `statistics` / `decimal`

```python
# ---- 本段讲：math / statistics / decimal 三组数值工具 ----
# math = java.lang.Math；statistics 是简易统计函数（不必引第三方）；
# decimal = BigDecimal，凡是钱、税率、手续费都必须用它，不能用 float。
import math
import statistics
from decimal import Decimal, ROUND_HALF_UP   # 导入十进制类型和「四舍五入」舍入模式

# math
# 这些函数都返回 float，除了 ceil/floor 返回 int。
math.sqrt(16)          # 4.0   平方根；负数会抛 ValueError（不是 NaN）
math.ceil(3.2)         # 4     向上取整
math.floor(3.8)        # 3     向下取整
math.isclose(0.1+0.2, 0.3)      # True  ← 浮点比较用这个；默认相对误差 1e-9
math.inf, -math.inf    # 正无穷、负无穷；可用于「初始最小值」等场景
math.pi, math.e        # 圆周率、自然常数，都是 float 常量

# statistics
# 输入是可迭代的数字序列，返回 float；样本量大时性能一般，适合脚本和小数据。
statistics.mean([1,2,3])         # 2     算术平均
statistics.median([1,2,3,4])     # 2.5   中位数；偶数个元素取中间两个的平均
statistics.stdev([1,2,3])        # 样本标准差；注意分母是 n-1（总体标准差用 pstdev）
statistics.quantiles([1,2,3,4,5], n=4)   # 四分位；n=4 表示切成 4 段，返回 3 个切分点

# decimal（金额必用）
# Decimal 用「十进制」存储，0.1 就是精确的 0.1，不会出现二进制浮点误差。
Decimal("0.1") + Decimal("0.2")                     # Decimal('0.3')
(Decimal("100") / Decimal("3")).quantize(Decimal("0.01"), rounding=ROUND_HALF_UP)   # 保留 2 位小数并四舍五入
```
