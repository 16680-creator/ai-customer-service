## 一、语法基础

```python
# ---- 本段讲：语法基础速查（变量 / 类型 / 数值 / 字符串 / 条件 / 循环 / match）----
# 都是可在 REPL 里逐行验证的片段；行尾 "→" 表示结果，"≈" 表示 Java 等价写法。
# 变量（无声明）
x = 1; s = "a"; b = True; n = None
# ↑ 赋值即定义，没有类型声明；分号可把多条语句写一行（Java 每句都必须分号）
a, b = b, a                    # 交换
# ↑ 右边先组成元组再解包给左边；Java 交换两个变量需要临时变量 tmp

# 类型
int   # 任意精度整数
# ↑ 没有溢出上限（Java 的 int 32 位、long 64 位，再大要 BigInteger）
float # 双精度（无 float32）
# ↑ 只有 64 位双精度；Java 的 float 在这里不存在，金额也别用它
str   # 不可变
# ↑ 字符串不可变，所有"修改"方法都返回新串（同 Java String）
bool  # True / False
# ↑ 首字母大写；Java 是 true/false
None  # 空值（用 is None 判断）
# ↑ 单例空值，判空用 is None（≈ == null），不要写 == None

# 数值
7 / 2      # 3.5      真除法
# ↑ / 永远返回 float；Java 的 int/int 会截断，是常见 bug 来源
7 // 2     # 3        整除
# ↑ 向下取整；负数时向负无穷（-7//2 == -4）
7 % 2      # 1
# ↑ 取模；结果符号跟随除数（与 Java 的 % 规则略有差异）
2 ** 10    # 1024
# ↑ 幂运算；Java 要 Math.pow(2,10) 且返回 double
abs(-3)    # 3
# ↑ 内置绝对值函数（Java 是 Math.abs）
round(3.567, 2)   # 3.57
# ↑ 保留 2 位小数，银行家舍入；金额场景请用 Decimal

# 字符串
s = "Hello"
len(s)              # 5
# ↑ len() 是内置函数；Java 是 s.length()
s[0], s[-1]         # 'H' 'o'
# ↑ 下标访问；负索引从末尾数（Java 不支持负索引）
s[1:4]              # 'ell'
# ↑ 切片，左闭右开 [1,4)，返回新字符串
s[::-1]             # 'olleH'
# ↑ 步长 -1 实现反转；Java 要 new StringBuilder(s).reverse()
s.upper(), s.lower()
# ↑ 转大写/小写，返回新串，原串不变
s.strip()           # 去空白
# ↑ 去掉首尾空白（含换行）；Java 的 trim() 只去 <= U+0020
s.replace("l","L")
# ↑ 全部字面量替换；Java 是 replace，正则替换用 replaceAll
s.split(",")        # → list
# ↑ 按分隔符切成列表；Java 的 split 返回数组
"-".join(["a","b"]) # 'a-b'
# ↑ 分隔符.join(可迭代)；注意调用方向与 Java 的 String.join(",", xs) 相反
"x" in s            # False
# ↑ 子串包含判断；Java 是 s.contains("x")
f"{x} {y:.2f}"      # f-string
# ↑ 格式化插值，{y:.2f} 保留两位；≈ String.format("%s %.2f", x, y)

# 条件
if a > b:
    ...
elif a == b:
    ...
else:
    ...
# ↑ 冒号 + 缩进代替大括号；elif 即 else if；... 是占位符，真代码要写具体语句
v = x if cond else y        # 三元（顺序与 Java 相反）
# ↑ 先写条件为真时的值；Java 写作 cond ? x : y

# 循环
for x in items: ...
# ↑ 直接迭代元素；≈ Java 的 for-each，但循环变量会留在外层作用域
for i, x in enumerate(items): ...
# ↑ 同时取下标和元素；Java 要自己维护 int i
for i in range(5): ...            # 0..4
# ↑ range(5) 生成 0~4（左闭右开）；≈ for(int i=0;i<5;i++)
for i in range(2, 10, 3): ...     # 2,5,8
# ↑ 起点、终点（不含）、步长
for k, v in d.items(): ...
# ↑ 遍历字典键值对；≈ for (Map.Entry<K,V> e : map.entrySet())
for a, b in zip(xs, ys): ...
# ↑ 并行遍历两个序列，按最短截断；Java 无直接等价物
while cond: ...
# ↑ 条件循环，写法同 Java，但没有 do-while
break / continue
# ↑ 跳出循环 / 跳过本次，语义同 Java（这行是语法速记，非可执行语句）
for ... else: ...                 # 无 break 才执行
# ↑ else 属于 for：循环正常跑完（没被 break）才执行，常用于"没找到"分支

# match（3.10+）
match cmd:
    case "start": ...
    case "a" | "b": ...
    case {"k": v}: ...
    case _: ...
# ↑ 结构化模式匹配：能解构 dict/list，比 Java 的 switch 强；_ 为兜底分支
```

## 二、容器

```python
# ---- 本段讲：容器速查（list / tuple / dict / set / 专用容器 / 推导式）----
# list≈ArrayList，tuple≈不可变 List，dict≈HashMap，set≈HashSet。
# list
lst = [1, 2, 3]
# ↑ 字面量建列表；≈ new ArrayList<>(List.of(1,2,3))，但 Python 列表可混装任意类型
lst.append(x); lst.extend(xs); lst.insert(i, x)
# ↑ 尾部追加 / 批量追加 / 指定位置插入；≈ add、addAll、add(i,x)；插入是 O(n)
lst.pop(); lst.pop(0); lst.remove(v)
# ↑ 弹尾部 / 弹指定下标 / 按值删除第一个匹配；下标越界抛 IndexError
lst.index(v); lst.count(v)
# ↑ 找下标（找不到抛 ValueError）/ 统计出现次数
lst.sort(); lst.reverse()
# ↑ 原地排序 / 原地反转；返回 None，别写成 lst = lst.sort()
len(lst); sum(lst); min(lst); max(lst)
# ↑ 内置统计函数；≈ size()、stream().mapToInt().sum()、Collections.min/max
x in lst
# ↑ 成员判断，O(n)；≈ contains
lst[a:b]; lst[::2]; lst[::-1]
# ↑ 切片（左闭右开）/ 隔一个取一个 / 反转；都返回新列表
lst.copy() / lst[:]
# ↑ 两种浅拷贝写法；嵌套对象仍是共享引用（深拷贝要 copy.deepcopy）
[x*2 for x in lst if x > 0]
# ↑ 列表推导式：映射 + 过滤；≈ stream().filter(x->x>0).map(x->x*2).toList()

# tuple
t = (1, 2); t = (1,)        # 单元素要逗号
# ↑ 元组不可变；(1,) 必须有逗号，否则 (1) 只是整数 1
a, b = t                    # 解包
# ↑ 按位置拆包；元素个数必须匹配，否则 ValueError

# dict
d = {"a": 1}
# ↑ 字典字面量；≈ new HashMap<>() 再 put，键必须可哈希
d["b"] = 2
# ↑ 赋值即插入/覆盖；≈ map.put("b", 2)
d.get(k); d.get(k, default)
# ↑ 安全取值，键不存在返回 None / 默认值；≈ Optional.map.orElse
d.setdefault(k, [])
# ↑ 键不存在才写入默认值并返回；常用于"分组累加"
d.pop(k); d.pop(k, None); del d[k]
# ↑ 弹出并返回 / 不存在时给默认值 / 直接删除（不存在会 KeyError）
d.keys(); d.values(); d.items()
# ↑ 视图对象；items() 产出 (键,值) 对，遍历时用
k in d
# ↑ 判断键是否存在，平均 O(1)；≈ containsKey
d1 | d2                     # 合并（3.9+）
# ↑ 返回新字典，右边覆盖左边；≈ putAll 但不可变
{k: v for k, v in pairs}
# ↑ 字典推导式；≈ stream().collect(toMap(...))

# set
s = {1, 2}; s = set()
# ↑ 字面量建集合 / 空集合必须用 set()（{} 是空字典！）
s.add(x); s.discard(x); s.remove(x)
# ↑ 添加 / 删除（不存在不报错）/ 删除（不存在抛 KeyError）
a | b; a & b; a - b; a ^ b
# ↑ 并集 / 交集 / 差集 / 对称差；≈ addAll、retainAll、removeAll
frozenset({1,2})            # 不可变，可作 key
# ↑ 不可变集合，因此可哈希，能当 dict 的键或 set 的元素

# 专用容器
from collections import Counter, defaultdict, deque, namedtuple
# ↑ 标准库 collections：计数器 / 带默认值字典 / 双端队列 / 具名元组
Counter("hello").most_common(2)
# ↑ 词频统计并取前 2 名；≈ stream().collect(groupingBy(counting()))
defaultdict(list)           # 免判空
# ↑ 访问不存在的键时自动创建 []（工厂函数），省掉 if key not in d 判断
deque([1,2], maxlen=3)
# ↑ 双端队列，超长自动从另一端淘汰；适合滑动窗口 / 固定容量缓存

# 推导式
[x for x in xs]                       # 映射/过滤
# ↑ 最基础的列表推导；结果是一个新 list
{k: v for k, v in pairs}              # 字典
# ↑ 字典推导；pairs 是 (键,值) 可迭代对象
{x % 3 for x in xs}                   # 集合
# ↑ 集合推导，自动去重
(x for x in xs)                       # 生成器（惰性）
# ↑ 圆括号得到生成器，惰性求值、只能遍历一次、省内存
[x for row in m for x in row]         # flatMap
# ↑ 双层 for 展平二维结构；≈ flatMap(row -> row.stream())
```

## 三、函数

```python
# ---- 本段讲：函数速查（参数形态 / 装饰器 / 生成器 / 缓存 / 内置高阶函数）----
# 参数声明顺序固定：位置参数 → 默认参数 → *args → 仅关键字参数 → **kwargs。
def f(a, b=2, *args, c, d=4, **kwargs):
    # a 必填；b 有默认值；*args 收集多余位置参数为元组；
    # c 必须用关键字传（keyword-only）；d 有默认值；**kwargs 收集多余关键字参数为字典
    return a

f(1, 2, 3, 4, c=5, e=6)
# ↑ 调用示例：a=1, b=2, args=(3,4), c=5, e 进 kwargs；Java 无同等语法，靠重载/Builder

lambda x: x * 2                       # 单表达式
# ↑ 匿名函数，只能写一个表达式；≈ x -> x * 2（Java 的 lambda）
def f(a, /, b, *, c): ...             # 位置only / 关键字only
# ↑ "/" 之前的参数只能按位置传，"*" 之后的只能按关键字传；... 为函数体占位

# 装饰器
import functools
# ↑ 提供 wraps / lru_cache 等工具
def deco(fn):
    # 接收被装饰函数，返回包装后的函数；≈ Java 的 AOP 环绕通知（但这里是显式包装）
    @functools.wraps(fn)
    # ↑ 保留原函数名/文档字符串等元数据，否则调试时全是 wrapper
    def wrapper(*a, **kw):
        # 用 *a/**kw 透传任意参数
        try:
            return fn(*a, **kw)
        finally:
            pass
            # ↑ finally 里放收尾逻辑（此处为空示意）；异常也会执行
    return wrapper

@deco
def g(): ...
# ↑ 语法糖：g = deco(g)；调用 g() 实际执行 wrapper

# 带参装饰器
def deco2(n=3):
    # 外层先收装饰器参数 n，返回真正的装饰器
    def _d(fn):
        @functools.wraps(fn)
        def w(*a, **kw): return fn(*a, **kw)
        # ↑ 这里可以读 n（闭包）；单行函数体直接 return
        return w
    return _d

@deco2(n=5)
def h(): ...
# ↑ 等价 h = deco2(n=5)(h)，三层嵌套是带参装饰器的标准写法

# 生成器
def gen(n):
    for i in range(n):
        yield i
        # ↑ yield 让函数变成生成器：每次产出后暂停，下次从暂停处继续；≈ Stream.iterate 的惰性版

# 缓存
@functools.lru_cache(maxsize=128)
# ↑ 缓存最近 128 个不同参数的结果；≈ Caffeine/Guava Cache
@functools.cache
# ↑ 无上限缓存（3.9+），等价 lru_cache(maxsize=None)

# 内置
map(f, xs); filter(p, xs); zip(xs, ys)
# ↑ 映射 / 过滤 / 拉链；都返回惰性迭代器，要列表得再套 list()
sorted(xs, key=fn, reverse=True)
# ↑ 返回新列表排序，key 指定比较键；≈ Comparator.comparing
any(...); all(...)
# ↑ 任一为真 / 全部为真；空序列时 any=False、all=True
sum(xs); min(xs, key=fn); max(...)
# ↑ 求和 / 按 key 取最小 / 取最大
enumerate(xs, start=1)
# ↑ 带下标迭代，下标从 1 开始；≈ IntStream.range + get
reversed(xs)
# ↑ 反向迭代器，不复制列表
```

## 四、类

```python
# ---- 本段讲：类速查（dataclass / 普通类 / 魔术方法 / 接口抽象）----
# Python 没有 getter/setter 样板代码；字段直接公开，需要时再用 @property 加逻辑。
from dataclasses import dataclass, field
# ↑ dataclass 自动生成 __init__/__repr__/__eq__；field 用于定制字段

@dataclass(frozen=True, slots=True)
# ↑ frozen=True 不可变（可哈希）；slots=True 省内存并禁止动态加属性
class Order:
    # ≈ record Order(String no, double amount, List<String> items)
    no: str
    # ↑ 类型注解 + 无默认值 ⇒ 构造时必填
    amount: float = 0.0
    # ↑ 有默认值的字段必须排在无默认值之后
    items: list[str] = field(default_factory=list)
    # ↑ 可变默认值必须用 factory，直接写 [] 会被所有实例共享

    def total(self) -> float:
        # 普通实例方法，第一个参数 self 相当于 Java 的 this
        return self.amount

class Plain:
    class_var = 0                    # 类属性（共享）
    # ↑ 定义在类体里、方法外；所有实例共享，≈ static 字段

    def __init__(self, x):
        # 构造器；≈ public Plain(int x)
        self.x = x                   # 实例属性
        # ↑ self.x 必须在方法内赋值才会存在；≈ this.x = x

    def m(self): ...                 # 实例方法
    # ↑ 第一个参数 self 由解释器自动传入，调用时不用写
    @classmethod
    def create(cls, ...): ...        # 类方法（工厂）
    # ↑ 第一个参数是类本身 cls；≈ static 工厂方法，可被继承
    @staticmethod
    def util(): ...                  # 静态方法
    # ↑ 不接收 self/cls；≈ Java 的 static 方法
    @property
    def computed(self): ...          # 属性
    # ↑ 调用时不用加括号：obj.computed；≈ getter
    @computed.setter
    def computed(self, v): ...
    # ↑ 给属性赋值时触发；≈ setter

    def __repr__(self): ...          # 调试显示
    # ↑ repr(obj) / 交互式回显用；目标是"能重建对象"的字符串
    def __str__(self): ...           # print 显示
    # ↑ print(obj) / str(obj) 用；≈ toString()
    def __eq__(self, o): ...
    # ↑ == 运算符；≈ equals
    def __hash__(self): ...
    # ↑ hash(obj)；实现 __eq__ 后若要用作 set/dict 键必须同时实现
    def __len__(self): ...
    # ↑ len(obj)；让对象支持 len()
    def __getitem__(self, i): ...
    # ↑ obj[i] 下标访问；让对象像容器
    def __iter__(self): ...
    # ↑ for x in obj 迭代协议
    def __enter__/__exit__           # with
    # ↑ 上下文管理器协议：进入/退出 with 块时调用

# 接口
from typing import Protocol
# ↑ 结构化子类型（鸭子类型的静态检查版）；≈ Java interface，但无需显式 implements
class Repo(Protocol):
    def find(self, id: int): ...
    # ↑ 只声明签名；任何有 find 方法的类都自动满足该协议（mypy 检查）

from abc import ABC, abstractmethod
# ↑ 抽象基类方案；≈ abstract class，需要显式继承
class Base(ABC):
    @abstractmethod
    def m(self): ...
    # ↑ 子类必须实现，否则实例化报错
```

## 五、异常与文件

```python
# ---- 本段讲：异常 + 文件 + 路径 + JSON + 金额 的常用写法 ----
# 重点：没有 checked exception；所有文件操作都要显式 encoding="utf-8"。
try:
    ...
    # ↑ 可能抛异常的代码
except (A, B) as e:
    # ↑ 同时捕获多种异常，as e 绑定异常对象；≈ catch (A | B e)
    ...
except Exception as e:
    # ↑ 捕获所有常规异常（不要裸 except:，会连 KeyboardInterrupt 都吞掉）
    raise
    # ↑ 不带参数重新抛出，保留原始堆栈；≈ throw e
else:
    ...                 # 无异常才执行
    # ↑ try 成功后才跑，适合放"依赖成功结果"的后续逻辑
finally:
    ...
    # ↑ 无论是否异常都执行，用于释放资源

raise ValueError("msg")
# ↑ 主动抛异常；≈ throw new IllegalArgumentException("msg")
raise X("msg") from e   # 保留原因
# ↑ 异常链：新异常保留原始异常；≈ new X("msg", e)
raise X("msg") from None # 切断链
# ↑ 不暴露底层异常，常用于对外接口隐藏实现细节

class AppError(Exception):
    # 自定义异常；≈ class AppError extends RuntimeException
    def __init__(self, code, msg):
        super().__init__(msg)
        # ↑ 把 msg 交给基类，str(e) 才能拿到消息；≈ super(msg)
        self.code = code
        # ↑ 额外挂业务码

# 上下文管理器
with open(p, encoding="utf-8") as f:
    ...
    # ↑ with 结束自动关闭文件，异常也会关；≈ try-with-resources
    #   encoding 必须显式写，否则用系统默认编码（Windows 下常是 GBK，必踩坑）

from contextlib import contextmanager
# ↑ 用生成器快速实现上下文管理器
@contextmanager
def timer():
    import time; t = time.perf_counter()
    # ↑ 进入 with 前的准备；分号可写多条（仅演示）
    try:
        yield
        # ↑ yield 处暂停，把控制权交给 with 块；块结束回来继续执行
    finally:
        print(time.perf_counter() - t)
        # ↑ 收尾逻辑放 finally，异常时也能打印耗时

# pathlib
from pathlib import Path
# ↑ 面向对象的路径库，跨平台；≈ java.nio.file.Path
p = Path("a") / "b.txt"
# ↑ "/" 运算符拼接路径，不用手写分隔符；≈ Paths.get("a","b.txt")
p.name; p.stem; p.suffix; p.parent
# ↑ 文件名 / 去扩展名 / 扩展名 / 父目录
p.exists(); p.is_file(); p.is_dir()
# ↑ 存在性 / 是否文件 / 是否目录；≈ Files.exists / isRegularFile / isDirectory
p.read_text(encoding="utf-8")
# ↑ 一次读全部文本；≈ Files.readString
p.write_text(s, encoding="utf-8")
# ↑ 一次写全部文本（覆盖）；≈ Files.writeString
p.read_bytes(); p.write_bytes(b)
# ↑ 读写二进制；≈ Files.readAllBytes / write
p.mkdir(parents=True, exist_ok=True)
# ↑ 递归建目录且已存在不报错；≈ Files.createDirectories
p.unlink(missing_ok=True)
# ↑ 删除文件，不存在也不报错；≈ Files.deleteIfExists
Path(".").rglob("*.py")
# ↑ 递归匹配所有 .py；≈ Files.walk + filter
p.resolve()
# ↑ 转成绝对路径并消解 .. 和符号链接；≈ toAbsolutePath.normalize

# json
import json
# ↑ 标准库 JSON；≈ Jackson ObjectMapper
json.dumps(d, ensure_ascii=False, indent=2)
# ↑ 对象 → JSON 字符串；ensure_ascii=False 保留中文，indent 美化
json.loads(s)
# ↑ JSON 字符串 → 对象（dict/list）
json.dump(d, f, ensure_ascii=False)
# ↑ 直接写进文件对象；省一次拼字符串
json.load(f)
# ↑ 从文件对象读取并解析

# 金额
from decimal import Decimal, ROUND_HALF_UP
# ↑ 十进制精确运算；≈ BigDecimal + RoundingMode.HALF_UP
(Decimal("100") / Decimal("3")).quantize(Decimal("0.01"), rounding=ROUND_HALF_UP)
# ↑ 除法后量化到 2 位小数并按四舍五入；金额禁用 float
```

## 六、并发

```python
# ---- 本段讲：并发速查（asyncio / 限流 / 线程池 / 进程池 / 队列）----
# 决策口诀：IO 密集 → 协程或线程池；CPU 密集 → 进程池（GIL 限制）。
# IO 密集 → asyncio
import asyncio
# ↑ 标准库事件循环；≈ CompletableFuture + 事件循环（但语法更接近 async/await）

async def f():
    # async def 定义协程函数；调用它只创建协程对象，不会立即执行
    await asyncio.sleep(1)
    # ↑ await 处让出事件循环；≈ Thread.sleep(1000) 但不会占线程

async def main():
    await asyncio.gather(f(), f())              # 并发
    # ↑ 同时启动两个协程并等全部完成；≈ CompletableFuture.allOf
    async with asyncio.TaskGroup() as tg:       # 3.11+
        tg.create_task(f())
        # ↑ 结构化并发：块结束时自动等待所有任务；异常会取消同组其他任务
    await asyncio.wait_for(f(), timeout=1)
    # ↑ 给协程加超时，超时抛 TimeoutError；≈ orTimeout
    await asyncio.to_thread(blocking_fn, arg)   # 阻塞调用隔离 ← 必须
    # ↑ 把同步阻塞函数丢到线程池执行；否则会卡死事件循环

asyncio.run(main())
# ↑ 程序入口：创建事件循环、跑 main、收尾；≈ SpringApplication.run 的启动作用

# 限流
sem = asyncio.Semaphore(10)
# ↑ 信号量，限制并发数；≈ java.util.concurrent.Semaphore
async with sem: ...
# ↑ 进入时 acquire、退出时 release；超出 10 个的协程在此排队

# 线程池（IO 密集 + 同步库）
from concurrent.futures import ThreadPoolExecutor, as_completed
# ↑ 线程池 / 完成顺序迭代；≈ ExecutorService + CompletionService
with ThreadPoolExecutor(max_workers=10) as pool:
    # ↑ with 结束时自动 shutdown；max_workers 是并发线程数
    list(pool.map(fn, xs))
    # ↑ 批量提交并按输入顺序取结果；≈ invokeAll
    for fut in as_completed(fs): fut.result()
    # ↑ 按"谁先完成"处理结果，fut.result() 取返回值或抛异常

# 进程池（CPU 密集）
from concurrent.futures import ProcessPoolExecutor
# ↑ 多进程绕过 GIL；≈ fork/join 但接口一致
if __name__ == "__main__":
    # ↑ Windows 下必须放在这个守卫里，否则子进程会重复导入模块导致递归
    with ProcessPoolExecutor() as pool:
        pool.map(fn, xs)       # fn 必须是模块级函数
        # ↑ 函数和参数都要可 pickle；lambda / 局部函数不能用于进程池

# 队列
import queue
# ↑ 线程安全队列；≈ BlockingQueue
q = queue.Queue(maxsize=100)
# ↑ 有界队列，满了 put 会阻塞（背压）
q.put(x); q.get(); q.task_done()
# ↑ 入队 / 出队 / 标记任务完成（配合 q.join() 等待全部处理完）
```

**决策**：IO 密集 → asyncio / 线程池；CPU 密集 → 进程池。
