# 25 · 可观测性实战：日志、监控、链路追踪

> 让生产环境"会说话"。日志、指标、链路追踪是诊断问题的三要素。本篇对标你的 Spring Boot 项目中的 SLF4J + Prometheus + Zipkin，用 Python 实现同等效果的可观测性。

## 一、结构化日志：从 `print()` 到生产级日志

### 为什么不能用 `print()`

```python
# ❌ 调试可以，生产不行
print(f"用户 {user_id} 登录失败")

# 问题：
# 1. 无法按级别过滤（ERROR/WARN/INFO/DEBUG）
# 2. 无法追踪请求链路（trace_id）
# 3. 无法导出到日志系统（ELK/Loki）
# 4. 时间戳、来源文件、行号丢失
```

### Python 的 `logging` 模块（对标 SLF4J）

```python
import logging
from pythonjsonlogger import jsonlogger

# ① 基础配置
logger = logging.getLogger(__name__)
logger.setLevel(logging.DEBUG)

# ② JSON 格式（便于日志系统解析）
handler = logging.StreamHandler()
formatter = jsonlogger.JsonFormatter()
handler.setFormatter(formatter)
logger.addHandler(handler)

# ③ 使用
logger.info("用户登录", extra={"user_id": 123})
# → {"message": "用户登录", "user_id": 123, "levelname": "INFO", "name": "__main__"}
```

### FastAPI 中集成结构化日志

```python
# app/core/logging_config.py
import logging
import json
from datetime import datetime

class JSONFormatter(logging.Formatter):
    def format(self, record):
        log_data = {
            "timestamp": datetime.utcnow().isoformat(),
            "level": record.levelname,
            "logger": record.name,
            "message": record.getMessage(),
            "module": record.module,
            "lineno": record.lineno,
        }
        
        # 附加 extra 字段
        if hasattr(record, "trace_id"):
            log_data["trace_id"] = record.trace_id
        if hasattr(record, "user_id"):
            log_data["user_id"] = record.user_id
            
        return json.dumps(log_data, ensure_ascii=False)


def setup_logging(level=logging.INFO):
    logger = logging.getLogger()
    logger.setLevel(level)
    
    handler = logging.StreamHandler()
    handler.setFormatter(JSONFormatter())
    logger.addHandler(handler)
    
    return logger


# app/main.py
from fastapi import FastAPI, Request
from app.core.logging_config import setup_logging
import logging
import uuid

setup_logging()
logger = logging.getLogger(__name__)

app = FastAPI()

# 中间件：给每个请求分配 trace_id（对标 Spring Cloud Sleuth）
@app.middleware("http")
async def trace_id_middleware(request: Request, call_next):
    trace_id = request.headers.get("X-Trace-Id") or str(uuid.uuid4())
    
    # 绑定到上下文（后续所有日志自动包含）
    ctx_vars = {"trace_id": trace_id}
    
    # 手动通过 extra 传递（Python 3.12 前的方案）
    # 更好的办法见下面的 contextvars
    
    response = await call_next(request)
    response.headers["X-Trace-Id"] = trace_id
    return response
```

### 用 `contextvars` 简化链路传递

```python
# app/core/context.py
from contextvars import ContextVar

trace_id_var = ContextVar("trace_id", default=None)
user_id_var = ContextVar("user_id", default=None)

def get_trace_id():
    return trace_id_var.get()

def set_trace_id(trace_id):
    return trace_id_var.set(trace_id)


# app/core/logging_config.py（改进）
import logging
from app.core.context import get_trace_id, get_user_id

class ContextAwareFormatter(logging.Formatter):
    def format(self, record):
        record.trace_id = get_trace_id()
        record.user_id = get_user_id()
        return super().format(record)


# 使用
logger.info("订单创建", extra={"order_id": "NO001"})
# → {"trace_id": "abc123", "user_id": 42, "order_id": "NO001", ...}
```

### 对照 Spring Boot 的 MDC（Mapped Diagnostic Context）

```java
// Java：用 MDC 绑定上下文
MDC.put("traceId", traceId);
logger.info("订单创建");
// → traceId=abc123 订单创建

MDC.clear();  // 请求结束清理
```

```python
# Python：用 contextvars 替代（更现代）
trace_id_var.set(trace_id)
logger.info("订单创建")
# trace_id 自动从 contextvars 读取，不需要 extra
```

## 二、指标暴露：Prometheus（对标 Micrometer）

### 场景：追踪 AI 对话的响应时间和调用计数

```python
# app/core/metrics.py
from prometheus_client import Counter, Histogram, Gauge, start_http_server
import time

# ① 调用计数（总次数）
chat_requests_total = Counter(
    "chat_requests_total",
    "总对话请求数",
    ["model", "status"]  # 维度：模型、状态
)

# ② 响应时间分布（直方图）
chat_duration_seconds = Histogram(
    "chat_duration_seconds",
    "对话响应耗时(秒)",
    ["model"],
    buckets=(0.1, 0.5, 1.0, 2.0, 5.0, 10.0)
)

# ③ 当前活跃请求数（仪表）
active_chat_requests = Gauge(
    "active_chat_requests",
    "当前处理中的对话"
)

# ④ 令牌使用量（计数器）
tokens_used = Counter(
    "tokens_used_total",
    "累计令牌消耗",
    ["model"]
)
```

### 在服务中埋点

```python
# app/services/chat_service.py
from app.core.metrics import *
import logging

logger = logging.getLogger(__name__)

class ChatService:
    async def reply(self, req: ChatRequest) -> ChatResponse:
        start = time.perf_counter()
        active_chat_requests.inc()
        
        try:
            text = await self.llm.chat(req.history, req.message)
            status = "success"
            return ChatResponse(session_id=req.session_id, reply=text, model=self.llm.settings.llm_model)
        except Exception as e:
            status = "error"
            logger.error("对话失败", extra={"error": str(e)})
            raise
        finally:
            active_chat_requests.dec()
            duration = time.perf_counter() - start
            
            # 记录指标
            chat_requests_total.labels(model=self.llm.settings.llm_model, status=status).inc()
            chat_duration_seconds.labels(model=self.llm.settings.llm_model).observe(duration)
            tokens_used.labels(model=self.llm.settings.llm_model).inc(
                int(len(req.message) / 4)  # 简化的令牌估算
            )
```

### 暴露 Prometheus 端点

```python

from prometheus_client import start_http_server

def create_app() -> FastAPI:
    # 在 8001 端口启动 Prometheus metrics 服务器
    start_http_server(8001)
    
    app = FastAPI(...)
    return app
```

```

# HELP chat_duration_seconds 对话响应耗时(秒)
```

### 对照 Spring Boot 的 Micrometer

```java
// Java
MeterRegistry registry = ...;
registry.counter("chat_requests_total", "model", "gpt-4", "status", "success").increment();
Timer.Sample sample = Timer.start(registry);
// 业务代码
sample.stop(Timer.builder("chat_duration_seconds").publishPercentiles(0.5, 0.95).register(registry));
```

```python
# Python
chat_requests_total.labels(model="gpt-4", status="success").inc()
chat_duration_seconds.labels(model="gpt-4").observe(duration)
```

**核心差异**：Python 更简洁，直接在变量上 `.inc()` / `.observe()`，无需 registry 中间层。

## 三、链路追踪：OpenTelemetry（对标 Zipkin/Jaeger）

### 场景：追踪一个请求跨多个微服务的调用链

```python
# app/core/tracing.py
from opentelemetry import trace, metrics
from opentelemetry.sdk.trace import TracerProvider
from opentelemetry.sdk.trace.export import BatchSpanProcessor
from opentelemetry.exporter.jaeger.thrift import JaegerExporter
from opentelemetry.instrumentation.fastapi import FastAPIInstrumentor
from opentelemetry.instrumentation.requests import RequestsInstrumentor
from opentelemetry.instrumentation.sqlalchemy import SQLAlchemyInstrumentor

def init_tracing():
    # 配置 Jaeger 导出器
    jaeger_exporter = JaegerExporter(
        agent_host_name="localhost",
        agent_port=6831,  # Jaeger agent 端口
    )
    
    # 创建 tracer provider
    trace_provider = TracerProvider()
    trace_provider.add_span_processor(BatchSpanProcessor(jaeger_exporter))
    trace.set_tracer_provider(trace_provider)
    
    # 自动埋点：FastAPI、HTTP 客户端、数据库
    FastAPIInstrumentor.instrument_app(app)
    RequestsInstrumentor().instrument()
    SQLAlchemyInstrumentor().instrument()


# app/main.py
from app.core.tracing import init_tracing

app = FastAPI()
init_tracing()
```

### 手动创建 Span（追踪自定义操作）

```python
# app/services/chat_service.py
from opentelemetry import trace

tracer = trace.get_tracer(__name__)

class ChatService:
    async def reply(self, req: ChatRequest) -> ChatResponse:
        # 创建一个 span（开启一个追踪单元）
        with tracer.start_as_current_span("chat_reply") as span:
            span.set_attribute("session_id", req.session_id)
            span.set_attribute("message_length", len(req.message))
            
            # 子 span：调用 LLM
            with tracer.start_as_current_span("llm_call") as llm_span:
                llm_span.set_attribute("model", self.llm.settings.llm_model)
                text = await self.llm.chat(req.history, req.message)
            
            return ChatResponse(...)
```

### 与 Java 侧 Spring Cloud Sleuth 对比

| 维度 | Java | Python |
|------|------|--------|
| trace_id 自动传播 | ✅ Sleuth 自动注入请求头 | ✅ OpenTelemetry 自动提取/注入 |
| Span 创建 | `@Span` 注解 + 方法名自动 | `tracer.start_as_current_span()` 手动 |
| 日志绑定 | MDC + SLF4J 自动 | contextvars + 日志 formatter 手动 |
| 导出后端 | Zipkin / Jaeger / Datadog | 都支持（更模块化） |

## 四、三合一：整合日志、指标、链路

### 端到端的生产配置示例

```python
# app/core/observability.py
import logging
import json
from datetime import datetime
from pythonjsonlogger import jsonlogger
from opentelemetry import trace
from app.core.context import get_trace_id

class CorrelatedJSONFormatter(jsonlogger.JsonFormatter):
    """同时输出 trace_id，便于与 Jaeger 关联"""
    
    def add_fields(self, log_record, record, message_dict):
        super().add_fields(log_record, record, message_dict)
        
        # 添加 trace_id（从 contextvars 读）
        trace_id = get_trace_id()
        if trace_id:
            log_record["trace_id"] = trace_id
        
        # 添加 span 上下文
        span = trace.get_current_span()
        if span and span.is_recording():
            log_record["span_id"] = hex(span.get_span_context().span_id)[2:]
            log_record["trace_id"] = hex(span.get_span_context().trace_id)[2:]


# 实际使用
logger = logging.getLogger(__name__)
handler = logging.StreamHandler()
handler.setFormatter(CorrelatedJSONFormatter())
logger.addHandler(handler)

logger.info("订单创建成功", extra={
    "order_id": "NO001",
    "amount": 10000,
})

# 输出包含 trace_id、span_id，可在 Jaeger UI 中点击日志直接跳转
```

## 五、故障排查：用可观测性定位问题

### 案例 1：某个用户的请求总是超时

```
① 从 Grafana 看到 P95 耗时突增到 5s
② 在 Jaeger 里查询 trace_id（从日志找到）
③ 发现卡在"LLM 调用"这一步
④ 切换模型或降级处理
```

### 案例 2：定位内存泄漏

```python
# app/core/metrics.py 新增
memory_usage_bytes = Gauge("process_memory_bytes", "进程内存占用")

# 后台任务定期上报
@app.on_event("startup")
async def start_memory_monitoring():
    import psutil
    import asyncio
    
    async def monitor():
        process = psutil.Process()
        while True:
            memory_usage_bytes.set(process.memory_info().rss)
            await asyncio.sleep(30)
    
    asyncio.create_task(monitor())
```

在 Grafana 看到内存持续增长 → 在 Jaeger 里查"哪个操作后内存增加" → 定位到代码。

## 八、与 Spring Boot 的对比总结

| 能力 | Spring Boot + Sleuth + Micrometer | FastAPI + OpenTelemetry + prometheus_client |
|------|-------|-------|
| 日志链路传递 | MDC 自动注入 | contextvars 手动（更灵活） |
| 指标 API | registry.counter().increment() | counter.labels(...).inc() |
| 分布式链路 | Sleuth + Zipkin | OpenTelemetry + Jaeger |
| 配置复杂度 | 高（大量 properties） | 低（代码式配置） |
| 性能开销 | 中等 | 更低（更轻量） |
