   29 | /**
   30 |  * Implementation of {@link Evaluator} used to evaluate the factual accuracy of Large
   31 |  * Language Model (LLM) responses against provided context.
   32 |  * <p>
   33 |  * This evaluator addresses a specific type of potential error in LLM outputs known as
   34 |  * "hallucination" in the context of grounded factuality. It verifies whether a given
   35 |  * statement (the "claim") is logically supported by a provided context (the "document").
   36 |  * <p>
   37 |  * Key concepts: - Document: The context or grounding information against which the claim
   38 |  * is checked. - Claim: The statement to be verified against the document.
   39 |  * <p>
   40 |  * The evaluator uses a prompt-based approach with a separate, typically smaller and more
   41 |  * efficient LLM to perform the fact-checking. This design choice allows for
   42 |  * cost-effective and rapid verification, which is crucial when evaluating longer LLM
   43 |  * outputs that may require multiple verification steps.
   44 |  * <p>
   45 |  * Implementation note: For efficient and accurate fact-checking, consider using
   46 |  * specialized models like Bespoke-Minicheck, a grounded factuality checking model
   47 |  * developed by Bespoke Labs and available in Ollama. Such models are specifically
   48 |  * designed to fact-check responses generated by other models, helping to detect and
   49 |  * reduce hallucinations. For more information, see:
   50 |  * <a href="https://ollama.com/blog/reduce-hallucinations-with-bespoke-minicheck">Reduce
   51 |  * Hallucinations with Bespoke-Minicheck</a> and the research paper:
   52 |  * <a href="https://arxiv.org/pdf/2404.10774v1">MiniCheck: An Efficient Method for LLM
   53 |  * Hallucination Detection</a>
   54 |  * <p>
   55 |  * Note: This evaluator is specifically designed to fact-check statements against given
   56 |  * information. It's not meant for other types of accuracy tests, like quizzing an AI on
   57 |  * obscure facts without giving it any reference material to work with (so-called 'closed
   58 |  * book' scenarios).
   59 |  * <p>
   60 |  * The evaluation process aims to determine if the claim is supported by the document,
   61 |  * returning a boolean result indicating whether the fact-check passed or failed.
   62 |  *
   63 |  * @author Eddú Meléndez
   64 |  * @author Mark Pollack
   65 |  * @author guan xu
   66 |  * @author Yanming Zhou
   67 |  * @see Evaluator
   68 |  * @see EvaluationRequest
   69 |  * @see EvaluationResponse
   70 |  * @since 1.0.0
   71 |  */
   72 | public class FactCheckingEvaluator implements Evaluator {

   74 | 	private static final String DEFAULT_EVALUATION_PROMPT_TEXT = """
   75 | 				Evaluate whether or not the following claim is supported by the provided document.
   76 | 				Respond with "yes" if the claim is supported, or "no" if it is not.
   77 | 
   78 | 				Document:
   79 | 				{document}
   80 | 
   81 | 				Claim:
   82 | 				{claim}
   83 | 			""";

   86 | 				Document:
   87 | 				{document}
   88 | 
   89 | 				Claim:
   90 | 				{claim}
   91 | 			""";

   94 | 
   95 | 	private final String evaluationPrompt;

   97 | 	/**
   98 | 	 * Constructs a new FactCheckingEvaluator with the provided ChatClient.Builder and
   99 | 	 * evaluation prompt.
  100 | 	 * @param chatClientBuilder The builder for the ChatClient used to perform the
  101 | 	 * evaluation
  102 | 	 * @param evaluationPrompt The prompt text to use for evaluation
  103 | 	 */
  104 | 	protected FactCheckingEvaluator(ChatClient.Builder chatClientBuilder, @Nullable String evaluationPrompt) {
  105 | 		Assert.notNull(chatClientBuilder, "chatClientBuilder cannot be null");
  106 | 		this.chatClientBuilder = chatClientBuilder;
  107 | 		this.evaluationPrompt = evaluationPrompt != null ? evaluationPrompt : DEFAULT_EVALUATION_PROMPT_TEXT;
  108 | 	}
  109 | 
  110 | 	/**
  111 | 	 * Creates a FactCheckingEvaluator configured for use with the Bespoke Minicheck
  112 | 	 * model.
  113 | 	 * @param chatClientBuilder The builder for the ChatClient used to perform the
  114 | 	 * evaluation
  115 | 	 * @return A FactCheckingEvaluator configured for Bespoke Minicheck
  116 | 	 */
  117 | 	public static FactCheckingEvaluator forBespokeMinicheck(ChatClient.Builder chatClientBuilder) {
  118 | 		return FactCheckingEvaluator.builder(chatClientBuilder)
  119 | 			.evaluationPrompt(BESPOKE_EVALUATION_PROMPT_TEXT)
  120 | 			.build();
  121 | 	}
  122 | 
  123 | 	/**
  124 | 	 * Evaluates whether the response content in the EvaluationRequest is factually
  125 | 	 * supported by the context provided in the same request.
  126 | 	 * @param evaluationRequest The request containing the response to be evaluated and
  127 | 	 * the supporting context
  128 | 	 * @return An EvaluationResponse indicating whether the claim is supported by the
  129 | 	 * document
  130 | 	 */
  131 | 	@Override
  132 | 	public EvaluationResponse evaluate(EvaluationRequest evaluationRequest) {
  133 | 		var response = evaluationRequest.getResponseContent();
  134 | 		var context = doGetSupportingData(evaluationRequest);
  135 | 
  136 | 		String evaluationResponse = this.chatClientBuilder.build()
  137 | 			.prompt()
  138 | 			.user(userSpec -> userSpec.text(this.evaluationPrompt).param("document", context).param("claim", response))
  139 | 			.call()
  140 | 			.content();
  141 | 
  142 | 		String normalizedResponse = (evaluationResponse != null) ? evaluationResponse.strip() : "";
  143 | 		boolean passing = "yes".equalsIgnoreCase(normalizedResponse);
  144 | 		return new EvaluationResponse(passing, "", Collections.emptyMap());
  145 | 	}

  148 | 		return new FactCheckingEvaluator.Builder().chatClientBuilder(chatClientBuilder);
  149 | 	}

  158 | 		}

  162 | 			return this;
  163 | 		}
  164 |

  167 | 			return this;
  168 | 		}
  169 | 
  170 | 		public FactCheckingEvaluator build() {
  171 | 			Assert.state(this.chatClientBuilder != null, "ChatClientBuilder cannot be null");
  172 | 			Assert.state(this.evaluationPrompt != null, "EvaluationPrompt cannot be null");
  173 | 			return new FactCheckingEvaluator(this.chatClientBuilder, this.evaluationPrompt);
  174 | 		}
  175 | 
  176 | 	}
