   74 | 	private void init() {
   75 | 		/*
   76 | 		 * Using a volatile invariant allows us to initialize the FastClass and
   77 | 		 * method index pairs atomically.
   78 | 		 *
   79 | 		 * Double-checked locking is safe with volatile in Java 5.  Before 1.5 this
   80 | 		 * code could allow fastClassInfo to be instantiated more than once, which
   81 | 		 * appears to be benign.
   82 | 		 */

   84 | 			synchronized (initLock) {
   85 | 				if (fastClassInfo == null) {
   86 | 					CreateInfo ci = createInfo;

   89 | 					fci.f1 = helper(ci, ci.c1);
   90 | 					fci.f2 = helper(ci, ci.c2);
   91 | 					fci.i1 = fci.f1.getIndex(sig1);
   92 | 					fci.i2 = fci.f2.getIndex(sig2);
   93 | 					fastClassInfo = fci;
   94 | 					createInfo = null;
   95 | 				}
   96 | 			}
   97 | 		}
   98 | 	}
